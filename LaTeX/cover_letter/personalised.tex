% Shift from current text position to left paper edge.
\newcommand*\ShiftToLeftPaperEdge{%
  \hspace*{-2\dimexpr 1in+\hoffset+\oddsidemargin\relax}%
}

% Shift from left paper edge back into the text area.
\newcommand*\ShiftBackToTextArea{%
  \hspace*{2\dimexpr 1in+\hoffset+\oddsidemargin\relax}%
}
